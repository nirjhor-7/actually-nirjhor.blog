    int num;

    int pCount = 0, iCount = 0;

    for (int i = 0; i < 15; i++) {
        scanf("%d", &num);

        if (num % 2 == 0) { // even

            if (pCount == 5) {

                    printf("par[%d] = %d\n", j, par[j]);

            }
        }
    }

    // Print remaining odds
    for (int j = 0; j < iCount; j++) {
        printf("impar[%d] = %d\n", j, impar[j]);
    }

    // Print remaining evens
    for (int j = 0; j < pCount; j++) {

#include <stdio.h>
#include <stdlib.h>

int main() {
    int N;
    scanf("%d", &N);

    long long *sheep = (long long*) malloc(N * sizeof(long long));
    for (int i = 0; i < N; i++) {
        scanf("%lld", &sheep[i]);
    }

    int *visited = (int*) calloc(N, sizeof(int));
    int pos = 0; // start at Star 1 (index 0)
    int attacked = 0;

    while (pos >= 0 && pos < N) {
        if (!visited[pos]) {
            attacked++;
            visited[pos] = 1;
        }

        long long before = sheep[pos]; // store before stealing
        if (sheep[pos] > 0) {
            sheep[pos]--; // steal one sheep
        }

        if (before % 2 == 1) {
            pos++; // odd → go right
        } else {
            pos--; // even → go left
        }
    }

    long long totalRemaining = 0;
    for (int i = 0; i < N; i++) {
        totalRemaining += sheep[i];
    }

    printf("%d %lld\n", attacked, totalRemaining);

    free(sheep);
    free(visited);

    return 0;
}
